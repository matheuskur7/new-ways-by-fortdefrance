import { StyleSheet, Text } from "@react-pdf/renderer";
import type { ServiceOrder, ServiceOrderExecuteItem } from "@/api/service-orders";
import type { ServiceOrderConfigType } from "@/api/service-order-types";
import { getDisabledCellsByCategory, getDisabledCellsForCategory } from "@/lib/service-order-config";
import type { PdfTheme } from "./service-order-pdf-document";
import { PdfTableCell, PdfTableFooter, PdfTableFooterCell, PdfTableHeader, PdfTableHeaderCell, PdfTableRow } from "./pdf-table";
import { PdfSection } from "./pdf-section";

type Props = { rows: number; serviceOrder: ServiceOrder; types: ServiceOrderConfigType[]; theme: PdfTheme };
type ExecuteTableStyles = ReturnType<typeof createStyles>;

export function ServiceOrderExecuteTable({ rows, serviceOrder, types, theme }: Props) {
  const items = serviceOrder.execute ?? [];
  const styles = createStyles(theme);
  const disabledCellsByCategory = getDisabledCellsByCategory(types);
  return (
    <PdfSection label="Executar" theme={theme}>
      <PdfTableHeader theme={theme}>
        <PdfTableHeaderCell style={styles.executeCell} theme={theme}>
          E
        </PdfTableHeaderCell>
        <PdfTableHeaderCell style={styles.descriptionCell} theme={theme}>
          Descrição peça / serviço
        </PdfTableHeaderCell>
        <PdfTableHeaderCell style={styles.statusCell} theme={theme}>
          Ped
        </PdfTableHeaderCell>
        <PdfTableHeaderCell style={styles.statusCell} theme={theme}>
          Sep
        </PdfTableHeaderCell>
        <PdfTableHeaderCell isLast style={styles.statusCell} theme={theme}>
          Sis
        </PdfTableHeaderCell>
      </PdfTableHeader>
      {Array.from({ length: rows }, (_, index) => (
        <ExecuteRow
          key={index}
          index={index}
          item={items[index]}
          disabledCellsByCategory={disabledCellsByCategory}
          styles={styles}
          theme={theme}
        />
      ))}
      <PdfTableFooter style={styles.footer} theme={theme}>
        <PdfTableFooterCell isLast style={styles.footerContent} theme={theme}>
          <Text style={styles.footerText}>
            E · Marcar suas iniciais quando concluir item · Ped · Pedido ao fornecedor · Sep · Separado no estoque · Sis · Lançado no
            sistema
          </Text>
        </PdfTableFooterCell>
      </PdfTableFooter>
    </PdfSection>
  );
}

function ExecuteRow({
  index,
  item,
  styles,
  theme,
  disabledCellsByCategory,
}: {
  index: number;
  item?: ServiceOrderExecuteItem;
  styles: ExecuteTableStyles;
  theme: PdfTheme;
  disabledCellsByCategory: Map<string, Set<string>>;
}) {
  const disabledCells = getDisabledCellsForCategory(disabledCellsByCategory, item?.category);
  return (
    <PdfTableRow striped={Boolean(index % 2)} style={styles.row} theme={theme}>
      <PdfTableCell variant={disabledCells?.has("execution") ? "disabled" : "default"} style={styles.executeCell} theme={theme} />
      <PdfTableCell style={styles.descriptionCell} theme={theme}>
        <Text style={styles.descriptionText}>{item?.description.toUpperCase()}</Text>
      </PdfTableCell>
      <PdfTableCell variant={disabledCells?.has("supplierOrder") ? "disabled" : "default"} style={styles.statusCell} theme={theme} />
      <PdfTableCell variant={disabledCells?.has("stockSeparation") ? "disabled" : "default"} style={styles.statusCell} theme={theme} />
      <PdfTableCell isLast variant={disabledCells?.has("systemEntry") ? "disabled" : "default"} style={styles.statusCell} theme={theme} />
    </PdfTableRow>
  );
}

function createStyles({ colors, layout }: PdfTheme) {
  const table = layout.reportTable;
  return StyleSheet.create({
    row: { height: table.rowHeight },
    executeCell: { width: table.columnWidth },
    descriptionCell: { flex: 1 },
    statusCell: { width: table.columnWidth },
    descriptionText: {
      color: colors.tableText,
      fontFamily: "Helvetica",
      fontSize: 8,
      paddingHorizontal: 2,
      paddingTop: 1.5,
      textTransform: "uppercase",
    },
    footer: { height: 12 },
    footerContent: { paddingHorizontal: 8 },
    footerText: { color: colors.tableText, fontFamily: "Helvetica-Bold", fontSize: 5, textTransform: "uppercase" },
  });
}
