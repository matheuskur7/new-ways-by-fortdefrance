"use client";

import dynamic from "next/dynamic";
import { useParams } from "next/navigation";
import { useEffect } from "react";
import { useGetCustomer } from "@/api/customers";
import { useServiceOrder } from "@/api/service-orders";
import { useServiceOrderReviewChecks } from "@/api/service-order-review-checks";
import { useServiceOrderReviewItems } from "@/api/service-order-review-items";
import { useServiceOrderCustomerReportTypes, useServiceOrderExecuteTypes } from "@/api/service-order-types";
import { ServiceOrderPdfDocument } from "../_components/service-order-pdf-document";

const BlobProvider = dynamic(() => import("@react-pdf/renderer").then((module) => module.BlobProvider), { ssr: false });

function PageMessage({ children }: { children: React.ReactNode }) {
  return <main className="flex h-dvh items-center justify-center text-sm text-muted-foreground">{children}</main>;
}

function OpenPdfInBrowser({ url }: { url: string }) {
  useEffect(() => {
    window.location.replace(url);
  }, [url]);

  return <PageMessage>Abrindo PDF...</PageMessage>;
}

export default function ServiceOrderPdfPage() {
  const { id } = useParams<{ id: string }>();
  const { data: serviceOrder, isLoading, isError } = useServiceOrder(id);
  const customerQuery = useGetCustomer(serviceOrder?.customerId ?? "");
  const reportTypes = useServiceOrderCustomerReportTypes();
  const executeTypes = useServiceOrderExecuteTypes();
  const reviewChecks = useServiceOrderReviewChecks();
  const reviewItems = useServiceOrderReviewItems();
  const isCustomerLoading = Boolean(serviceOrder?.customerId && customerQuery.isLoading);

  if (
    isLoading ||
    isCustomerLoading ||
    reportTypes.isLoading ||
    executeTypes.isLoading ||
    reviewChecks.isLoading ||
    reviewItems.isLoading
  ) {
    return <PageMessage>Carregando ordem de serviço...</PageMessage>;
  }

  if (
    isError ||
    customerQuery.isError ||
    reportTypes.isError ||
    executeTypes.isError ||
    reviewChecks.isError ||
    reviewItems.isError ||
    !serviceOrder
  ) {
    return <PageMessage>Não foi possível carregar a ordem de serviço.</PageMessage>;
  }

  return (
    <BlobProvider
      document={
        <ServiceOrderPdfDocument
          serviceOrder={serviceOrder}
          customer={customerQuery.data}
          customerReportTypes={reportTypes.data ?? []}
          executeTypes={executeTypes.data ?? []}
          reviewChecks={reviewChecks.data ?? []}
          reviewItems={reviewItems.data ?? []}
        />
      }
    >
      {({ error, url }) => {
        if (url) return <OpenPdfInBrowser url={url} />;
        if (error) return <PageMessage>Não foi possível gerar o PDF.</PageMessage>;

        return <PageMessage>Gerando PDF...</PageMessage>;
      }}
    </BlobProvider>
  );
}
