import type { ServiceOrder } from "@/api/service-orders";

  size = "sm",

  size?: ComponentProps<typeof Button>["size"];
}) {
  return (
    <Button asChild variant="outline" size={size}>

        <FilePdfIcon />
        Abrir PDF
      </Link>
    </Button>
  );
}
